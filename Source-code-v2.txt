#include <SPI.h>
#include <Adafruit_GFX.h>
#include <Adafruit_ST7735.h>
#include <PZEM004Tv40_R4.h>

// =====================================================
// PIN CONFIGURATION
// =====================================================

#define TFT_CS       53
#define TFT_DC       49
#define TFT_RST      48

#define RENT_SWITCH  22
#define RELAY_PIN    24
#define BUZZER_PIN   26

// If your TFT has a BL/LED pin connected to Mega,
// define it here. Otherwise leave as -1.
#define TFT_BACKLIGHT  -1

#define TARIFF 12.0f

// =====================================================
// TFT
// =====================================================

Adafruit_ST7735 tft = Adafruit_ST7735(
  TFT_CS,
  TFT_DC,
  TFT_RST
);

// =====================================================
// PZEM
// =====================================================

PZEM004Tv40_R4 pzem(&Serial2);

// =====================================================
// DISPLAY SIZE - LANDSCAPE
// =====================================================

#define SCREEN_W 160
#define SCREEN_H 128

// =====================================================
// RELAY LOGIC
// =====================================================

#define RELAY_ON  LOW
#define RELAY_OFF HIGH

// =====================================================
// STATES
// =====================================================

enum SystemState {
  STATE_WELCOME,
  STATE_READY,
  STATE_RENTING,
  STATE_SUMMARY,
  STATE_PAYMENT,
  STATE_WAIT,
  STATE_ERROR
};

SystemState state = STATE_WELCOME;

// =====================================================
// PZEM DATA
// =====================================================

float voltage   = 0;
float current   = 0;
float power     = 0;
float energy    = 0;
float frequency = 0;
float pf        = 0;

// =====================================================
// RENT SESSION
// =====================================================

float sessionStartEnergy = 0;
float sessionEndEnergy   = 0;
float sessionEnergy      = 0;
float sessionCost        = 0;

unsigned long rentalStartTime = 0;
unsigned long rentalEndTime   = 0;

// =====================================================
// STATE TIMING
// =====================================================

unsigned long stateStartTime = 0;
unsigned long lastPzemRead   = 0;
unsigned long lastDisplay    = 0;
unsigned long lastCountdown  = 0;

// =====================================================
// TIMINGS
// =====================================================

const unsigned long WELCOME_TIME = 5000;

// FULL 15 SECOND SUMMARY
const unsigned long SUMMARY_TIME = 15000;

// FULL 15 SECOND PAYMENT
const unsigned long PAYMENT_TIME = 15000;

const unsigned long PZEM_INTERVAL = 1000;

// =====================================================
// COLORS
// =====================================================

#define BLACK   ST77XX_BLACK
#define WHITE   ST77XX_WHITE
#define RED     ST77XX_RED
#define GREEN   ST77XX_GREEN
#define BLUE    ST77XX_BLUE
#define YELLOW  ST77XX_YELLOW
#define CYAN    ST77XX_CYAN

// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);

  Serial.println();
  Serial.println("SMART RENTAL POWER CONTROLLER");
  Serial.println("--------------------------------");

  // Switch
  pinMode(RENT_SWITCH, INPUT_PULLUP);

  // Relay
  pinMode(RELAY_PIN, OUTPUT);

  // VERY IMPORTANT:
  // Relay OFF during startup
  digitalWrite(RELAY_PIN, RELAY_OFF);

  // Buzzer
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  // Optional TFT backlight
  if (TFT_BACKLIGHT != -1) {
    pinMode(TFT_BACKLIGHT, OUTPUT);
    digitalWrite(TFT_BACKLIGHT, HIGH);
  }

  // TFT
  tft.initR(INITR_BLACKTAB);

  // Landscape
  tft.setRotation(1);

  tft.fillScreen(BLACK);

  // PZEM
  pzem.begin();

  state = STATE_WELCOME;

  stateStartTime = millis();

  drawWelcome();
}

// =====================================================
// MAIN LOOP
// =====================================================

void loop() {

  unsigned long now = millis();

  updatePZEM();

  switch (state) {

    case STATE_WELCOME:
      handleWelcome(now);
      break;

    case STATE_READY:
      handleReady();
      break;

    case STATE_RENTING:
      handleRenting(now);
      break;

    case STATE_SUMMARY:
      handleSummary(now);
      break;

    case STATE_PAYMENT:
      handlePayment(now);
      break;

    case STATE_WAIT:
      handleWait(now);
      break;

    case STATE_ERROR:
      handleError(now);
      break;
  }
}

// =====================================================
// PZEM UPDATE
// =====================================================

void updatePZEM() {

  if (millis() - lastPzemRead < PZEM_INTERVAL) {
    return;
  }

  lastPzemRead = millis();

  if (pzem.readAll()) {

    voltage   = pzem.getVoltage();
    current   = pzem.getCurrent();
    power     = pzem.getPower();
    energy    = pzem.getEnergy();
    frequency = pzem.getFrequency();
    pf        = pzem.getPowerFactor();

    Serial.print("V: ");
    Serial.print(voltage, 1);

    Serial.print("  I: ");
    Serial.print(current, 2);

    Serial.print("  P: ");
    Serial.print(power, 1);

    Serial.print("  E: ");
    Serial.print(energy, 3);

    Serial.print("  F: ");
    Serial.print(frequency, 1);

    Serial.print("  PF: ");
    Serial.println(pf, 2);

  } else {

    Serial.print("PZEM ERROR: ");
    Serial.println(pzem.getLastError());
  }
}

// =====================================================
// WELCOME SCREEN
// =====================================================

void drawWelcome() {

  tft.fillScreen(BLACK);

  tft.setTextColor(CYAN);
  tft.setTextSize(2);

  tft.setCursor(32, 15);
  tft.println("SMART");

  tft.setCursor(32, 40);
  tft.println("RENTAL");

  tft.setTextSize(1);

  tft.setTextColor(WHITE);

  tft.setCursor(35, 70);
  tft.println("POWER SYSTEM");

  tft.setCursor(35, 90);
  tft.println("Initializing");

  tft.setCursor(65, 108);
  tft.println("...");
}

void handleWelcome(unsigned long now) {

  static unsigned long lastAnimation = 0;
  static byte dots = 0;

  if (now - lastAnimation >= 400) {

    lastAnimation = now;

    tft.fillRect(60, 105, 40, 10, BLACK);

    tft.setCursor(60, 105);

    tft.setTextColor(YELLOW);

    for (byte i = 0; i < dots; i++) {
      tft.print(".");
    }

    dots++;

    if (dots > 4) {
      dots = 0;
    }
  }

  if (now - stateStartTime >= WELCOME_TIME) {

    state = STATE_READY;

    stateStartTime = now;

    drawReady();
  }
}

void handleReady() {

  if (millis() - lastDisplay >= 1000) {

    lastDisplay = millis();

    drawReady();
  }

  bool sw = digitalRead(RENT_SWITCH);

  if (sw == LOW) {

    startRental();
  }
}

// =====================================================
// START RENTAL
// =====================================================

void startRental() {

  sessionStartEnergy = energy;

  rentalStartTime = millis();

  // RELAY ON
  digitalWrite(RELAY_PIN, RELAY_ON);

  beep(1);

  state = STATE_RENTING;

  stateStartTime = millis();

  drawRenting();
}

// =====================================================
// RENTING SCREEN
// =====================================================

void drawRenting() {

  tft.fillScreen(BLACK);

  // Header
  tft.setTextSize(2);

  tft.setTextColor(GREEN);

  tft.setCursor(35, 3);
  tft.println("RENTING");

  tft.setTextSize(1);

  tft.setTextColor(WHITE);

  tft.setCursor(5, 27);
  tft.print("VOLTAGE : ");
  tft.print(voltage, 1);
  tft.println(" V");

  tft.setCursor(5, 42);
  tft.print("CURRENT : ");
  tft.print(current, 2);
  tft.println(" A");

  tft.setCursor(5, 57);
  tft.print("POWER   : ");
  tft.print(power, 1);
  tft.println(" W");

  if (usedEnergy < 0) {
    usedEnergy = 0;
  }

  float currentCost = usedEnergy * TARIFF;

  tft.setCursor(5, 72);

  tft.println(" kWh");

  tft.setCursor(5, 87);

  tft.print(currentCost, 2);

  tft.setCursor(90, 42);
  tft.print("F:");
  tft.print(frequency, 1);

  tft.setCursor(90, 57);
  tft.print("PF:");
  tft.print(pf, 2);

  // Separator
  tft.drawFastHLine(0, 102, SCREEN_W, WHITE);

  tft.setTextColor(YELLOW);

  tft.setCursor(13, 110);
  tft.println("SWITCH OFF TO FINISH");
}

void handleRenting(unsigned long now) {

  if (now - lastDisplay >= 1000) {

    lastDisplay = now;

    drawRenting();
  }

  bool sw = digitalRead(RENT_SWITCH);

  if (sw == HIGH) {

    finishRental();
  }
}

// =====================================================
// FINISH RENTAL
// =====================================================

void finishRental() {

  // Relay OFF immediately
  digitalWrite(RELAY_PIN, RELAY_OFF);

  rentalEndTime = millis();

  sessionEndEnergy = energy;

  sessionEnergy =
    sessionEndEnergy - sessionStartEnergy;

  if (sessionEnergy < 0) {
    sessionEnergy = 0;
  }

  sessionCost = sessionEnergy * TARIFF;

  beep(2);

  state = STATE_SUMMARY;

  stateStartTime = millis();

  lastCountdown = 0;

  drawSummary();
}

// =====================================================
// SUMMARY HANDLER
// =====================================================

void handleSummary(unsigned long now) {

  // Update countdown once per second
  if (now - lastCountdown >= 1000) {

    lastCountdown = now;

    unsigned long elapsed =
      now - stateStartTime;

    int remaining =
      15 - (elapsed / 1000);

    if (remaining < 0) {
      remaining = 0;
    }

    drawCountdown(remaining);
  }

  // User tries to start during summary
  if (digitalRead(RENT_SWITCH) == LOW) {

    beep(1);

    state = STATE_WAIT;

    stateStartTime = millis();

    drawWait();
    return;
  }

  // After 15 seconds
  if (now - stateStartTime >= SUMMARY_TIME) {

    state = STATE_PAYMENT;

    stateStartTime = millis();

    lastCountdown = 0;

    drawPayment();
  }
}

// =====================================================
// PAYMENT HANDLER
// =====================================================

void handlePayment(unsigned long now) {

  static unsigned long lastWarning = 0;

  // If user tries switch ON
  if (digitalRead(RENT_SWITCH) == LOW) {

    if (now - lastWarning >= 2000) {

      lastWarning = now;

      beep(1);

      drawWait();
    }
  }

  // Countdown
  if (now - lastCountdown >= 1000) {

    lastCountdown = now;

    unsigned long elapsed =
      now - stateStartTime;

    int remaining =
      15 - (elapsed / 1000);

    if (remaining < 0) {
      remaining = 0;
    }

    drawPaymentCountdown(remaining);
  }

  // After 15 seconds
  if (now - stateStartTime >= PAYMENT_TIME) {

    // Make sure relay remains OFF
    digitalWrite(RELAY_PIN, RELAY_OFF);

    state = STATE_READY;

    stateStartTime = now;

    drawReady();
  }
}

// =====================================================
// WAIT HANDLER
// =====================================================

void handleWait(unsigned long now) {

  // Relay MUST remain OFF
  digitalWrite(RELAY_PIN, RELAY_OFF);

  if (now - stateStartTime >= 15000) {

    state = STATE_READY;

    stateStartTime = now;

    drawReady();
  }
}

// =====================================================
// ERROR
// =====================================================

void showError(const char *message) {

  digitalWrite(RELAY_PIN, RELAY_OFF);

  tft.fillScreen(BLACK);

  tft.setTextColor(RED);

  tft.setTextSize(2);

  tft.setCursor(45, 10);
  tft.println("ERROR");

  tft.setTextSize(1);

  tft.setTextColor(WHITE);

  tft.setCursor(5, 50);

  tft.println(message);

  beep(3);

  state = STATE_ERROR;

  stateStartTime = millis();
}

// =====================================================
// ERROR HANDLER
// =====================================================

void handleError(unsigned long now) {

  digitalWrite(RELAY_PIN, RELAY_OFF);

  if (now - stateStartTime > 5000) {

    state = STATE_READY;

    drawReady();
  }
}

// =====================================================
// BUZZER
// =====================================================

void beep(byte count) {

  for (byte i = 0; i < count; i++) {

    digitalWrite(BUZZER_PIN, HIGH);

    delay(100);

    digitalWrite(BUZZER_PIN, LOW);

    delay(100);
  }
}

// =====================================================
// DUMMY QR
// =====================================================

void drawDummyQR(int x, int y) {

  const int module = 3;
  const int size = 21;

  tft.fillRect(
    x,
    y,
    size * module,
    size * module,
    WHITE
  );

  for (int row = 0; row < size; row++) {

    for (int col = 0; col < size; col++) {

      bool blackPixel = false;

      // Finder patterns
      if (
        (row < 7 && col < 7) ||
        (row < 7 && col >= size - 7) ||
        (row >= size - 7 && col < 7)
      ) {

        int fx;
        int fy;

        if (col < 7)
          fx = col;
        else if (col >= size - 7)
          fx = col - (size - 7);
        else
          fx = col;

        if (row < 7)
          fy = row;
        else
          fy = row - (size - 7);

        if (
          fx == 0 || fx == 6 ||
          fy == 0 || fy == 6 ||
          (fx >= 2 && fx <= 4 &&
           fy >= 2 && fy <= 4)
        ) {

          blackPixel = true;
        }

      } else {

        blackPixel =
          ((row * 13 + col * 7 + row * col) % 5) < 2;
      }

      if (blackPixel) {

        tft.fillRect(
          x + col * module,
          y + row * module,
          module,
          module,
          BLACK
        );
      }
    }
  }
}
