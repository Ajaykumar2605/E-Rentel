// =====================================================
// PIN CONFIGURATION
// =====================================================

#define TFT_CS       53
#define TFT_DC       49
#define TFT_RST      48

#define RENT_SWITCH  22
#define RELAY_PIN    24
#define BUZZER_PIN   26

#define TARIFF       12.0f

// =====================================================
// RELAY CONFIGURATION
// =====================================================
//
// YOUR RELAY:
// LOW  = ON
// HIGH = OFF
//
// =====================================================

#define RELAY_ON     LOW
#define RELAY_OFF    HIGH

// =====================================================
// TFT
// =====================================================

Adafruit_ST7735 tft = Adafruit_ST7735(
  TFT_CS,
  TFT_DC,
  TFT_RST
);

// =====================================================
// PZEM
// Mega Serial2:
// RX2 = 17
// TX2 = 16
// =====================================================

PZEM004Tv40_R4 pzem(&Serial2);

// =====================================================
// STATES
// =====================================================

enum SystemState {
  STATE_WELCOME,
  STATE_READY,
  STATE_RENTING,
  STATE_SUMMARY,
  STATE_PAYMENT,
  STATE_WAIT,
  STATE_ERROR
};

SystemState state = STATE_WELCOME;

// =====================================================
// PZEM DATA
// =====================================================

float voltage = 0;
float current = 0;
float power = 0;
float energy = 0;
float frequency = 0;
float pf = 0;

// =====================================================
// RENT SESSION
// =====================================================

float sessionStartEnergy = 0;
float sessionEndEnergy = 0;
float sessionEnergy = 0;
float sessionCost = 0;

unsigned long rentalStartTime = 0;
unsigned long rentalEndTime = 0;

unsigned long stateStartTime = 0;

// =====================================================
// TIMING
// =====================================================

const unsigned long WELCOME_TIME  = 5000UL;
const unsigned long SUMMARY_TIME  = 15000UL;
const unsigned long PAYMENT_TIME  = 15000UL;
const unsigned long WAIT_TIME     = 15000UL;
const unsigned long PZEM_INTERVAL = 1000UL;

unsigned long lastPzemRead = 0;

// =====================================================
// DISPLAY UPDATE
// =====================================================

unsigned long lastReadyDisplay = 0;
unsigned long lastRentalDisplay = 0;

// =====================================================
// WARNING BEEP
// =====================================================

unsigned long lastWarningBeep = 0;

// =====================================================
// COLORS
// =====================================================

#define BLACK   ST77XX_BLACK
#define WHITE   ST77XX_WHITE
#define RED     ST77XX_RED
#define GREEN   ST77XX_GREEN
#define BLUE    ST77XX_BLUE
#define YELLOW  ST77XX_YELLOW
#define CYAN    ST77XX_CYAN

// =====================================================
// RELAY FUNCTIONS
// =====================================================

// -----------------------------------------------------
// RELAY ON
// LOW = ON
// -----------------------------------------------------

void relayON() {
  digitalWrite(RELAY_PIN, LOW);
}

// -----------------------------------------------------
// RELAY OFF
// HIGH = OFF
// -----------------------------------------------------

void relayOFF() {
  digitalWrite(RELAY_PIN, HIGH);
}

// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);

  Serial.println();
  Serial.println("SMART RENTAL POWER CONTROLLER");
  Serial.println("--------------------------------");

  // ---------------------------------------------------
  // SWITCH
  // ---------------------------------------------------

  pinMode(RENT_SWITCH, INPUT_PULLUP);

  // ---------------------------------------------------
  // RELAY
  // ---------------------------------------------------

  pinMode(RELAY_PIN, OUTPUT);

  // IMPORTANT:
  // RELAY OFF = HIGH
  relayOFF();

  // ---------------------------------------------------
  // BUZZER
  // ---------------------------------------------------

  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  // ---------------------------------------------------
  // TFT
  // ---------------------------------------------------

  tft.initR(INITR_BLACKTAB);

  // 128 x 160 portrait
  tft.setRotation(0);

  tft.fillScreen(BLACK);

  // ---------------------------------------------------
  // PZEM
  // ---------------------------------------------------

  Serial2.begin(9600);
  pzem.begin();

  // ---------------------------------------------------
  // INITIAL STATE
  // ---------------------------------------------------

  state = STATE_WELCOME;
  stateStartTime = millis();

  drawWelcome();
}

// =====================================================
// MAIN LOOP
// =====================================================

void loop() {

  unsigned long now = millis();

  // Read PZEM
  updatePZEM();

  switch (state) {

    case STATE_WELCOME:
      handleWelcome(now);
      break;

    case STATE_READY:
      handleReady(now);
      break;

    case STATE_RENTING:
      handleRenting(now);
      break;

    case STATE_SUMMARY:
      handleSummary(now);
      break;

    case STATE_PAYMENT:
      handlePayment(now);
      break;

    case STATE_WAIT:
      handleWait(now);
      break;

    case STATE_ERROR:
      handleError(now);
      break;
  }
}

// =====================================================
// PZEM UPDATE
// =====================================================

void updatePZEM() {

  if (millis() - lastPzemRead < PZEM_INTERVAL) {
    return;
  }

  lastPzemRead = millis();

  if (pzem.readAll()) {

    voltage   = pzem.getVoltage();
    current   = pzem.getCurrent();
    power     = pzem.getPower();
    energy    = pzem.getEnergy();
    frequency = pzem.getFrequency();
    pf        = pzem.getPowerFactor();

    Serial.print("V: ");
    Serial.print(voltage);

    Serial.print("  I: ");
    Serial.print(current);

    Serial.print("  P: ");
    Serial.print(power);

    Serial.print("  E: ");
    Serial.print(energy);

    Serial.print("  F: ");
    Serial.print(frequency);

    Serial.print("  PF: ");
    Serial.println(pf);

  } else {

    Serial.print("PZEM ERROR: ");
    Serial.println(pzem.getLastError());
  }
}

// =====================================================
// WELCOME HANDLER
// =====================================================

void handleWelcome(unsigned long now) {

  static unsigned long lastAnimation = 0;
  static byte dots = 0;

  if (now - lastAnimation >= 400UL) {

    lastAnimation = now;

    tft.fillRect(20, 120, 100, 10, BLACK);

    tft.setCursor(20, 120);
    tft.setTextColor(YELLOW);
    tft.setTextSize(1);

    for (byte i = 0; i < dots; i++) {
      tft.print(".");
    }

    dots++;

    if (dots > 4) {
      dots = 0;
    }
  }

  if (now - stateStartTime >= WELCOME_TIME) {

    // SAFETY: relay OFF
    relayOFF();

    state = STATE_READY;
    stateStartTime = now;

    drawReady();
  }
}

// =====================================================
// READY HANDLER
// =====================================================

void handleReady(unsigned long now) {

  // ===================================================
  // RELAY MUST BE OFF
  // ===================================================

  relayOFF();

  // ---------------------------------------------------
  // UPDATE DISPLAY
  // ---------------------------------------------------

  if (now - lastReadyDisplay >= 1000UL) {

    lastReadyDisplay = now;

    drawReady();
  }

  // ---------------------------------------------------
  // SWITCH ON
  // ---------------------------------------------------

  if (digitalRead(RENT_SWITCH) == LOW) {

    delay(80);

    if (digitalRead(RENT_SWITCH) == LOW) {

      startRental();
    }
  }
}

// =====================================================
// START RENTAL
// =====================================================

void startRental() {

  // ---------------------------------------------------
  // CAPTURE START ENERGY
  // ---------------------------------------------------

  sessionStartEnergy = energy;

  sessionEndEnergy = sessionStartEnergy;

  sessionEnergy = 0;
  sessionCost = 0;

  rentalStartTime = millis();

  // ===================================================
  // RELAY ON
  // LOW = ON
  // ===================================================

  relayON();

  beep(1);

  state = STATE_RENTING;

  stateStartTime = millis();

  lastRentalDisplay = 0;

  drawRenting();
}

// =====================================================
// RENTING HANDLER
// =====================================================

void handleRenting(unsigned long now) {

  // ===================================================
  // RELAY MUST STAY ON
  // LOW = ON
  // ===================================================

  relayON();

  // ---------------------------------------------------
  // CALCULATE SESSION ENERGY
  // ---------------------------------------------------

  if (energy >= sessionStartEnergy) {

    sessionEnergy =
      energy - sessionStartEnergy;
  }
  else {

    sessionEnergy = 0;
  }

  sessionCost =
    sessionEnergy * TARIFF;

  // ---------------------------------------------------
  // UPDATE DISPLAY
  // ---------------------------------------------------

  if (now - lastRentalDisplay >= 1000UL) {

    lastRentalDisplay = now;

    drawRenting();
  }

  // ---------------------------------------------------
  // SWITCH OFF
  // ---------------------------------------------------

  if (digitalRead(RENT_SWITCH) == HIGH) {

    delay(80);

    if (digitalRead(RENT_SWITCH) == HIGH) {

      finishRental();
    }
  }
}

// =====================================================
// FINISH RENTAL
// =====================================================

void finishRental() {

  // ===================================================
  // IMMEDIATELY DISCONNECT LOAD
  // HIGH = RELAY OFF
  // ===================================================

  relayOFF();

  rentalEndTime = millis();

  // ---------------------------------------------------
  // FINAL ENERGY
  // ---------------------------------------------------

  sessionEndEnergy = energy;

  if (sessionEndEnergy >= sessionStartEnergy) {

    sessionEnergy =
      sessionEndEnergy - sessionStartEnergy;
  }
  else {

    sessionEnergy = 0;
  }

  // ---------------------------------------------------
  // FINAL COST
  // ---------------------------------------------------

  sessionCost =
    sessionEnergy * TARIFF;

  beep(2);

  // ---------------------------------------------------
  // SUMMARY
  // ---------------------------------------------------

  state = STATE_SUMMARY;

  stateStartTime = millis();

  drawSummary();
}

}

// =====================================================
// SUMMARY HANDLER
// =====================================================

void handleSummary(unsigned long now) {

  // ===================================================
  // RELAY OFF
  // ===================================================

  relayOFF();

  // ---------------------------------------------------
  // COUNTDOWN
  // ---------------------------------------------------

  drawSummaryCountdown();

  // ---------------------------------------------------
  // SWITCH ON DURING SUMMARY
  // ---------------------------------------------------

  if (digitalRead(RENT_SWITCH) == LOW) {

    // Do not start a new rental
    // Show WAIT state

    beep(1);

    state = STATE_WAIT;

    stateStartTime = now;

    drawWait();

    return;
  }

  // ---------------------------------------------------
  // SUMMARY FINISHED
  // ---------------------------------------------------

  if (now - stateStartTime >= SUMMARY_TIME) {

    state = STATE_PAYMENT;

    stateStartTime = now;

    drawPayment();
  }
}

// =====================================================
// PAYMENT HANDLER
// =====================================================

void handlePayment(unsigned long now) {

  // ===================================================
  // RELAY OFF
  // ===================================================

  relayOFF();

  // ---------------------------------------------------
  // COUNTDOWN
  // ---------------------------------------------------

  drawPaymentCountdown();

  // ---------------------------------------------------
  // SWITCH ON DURING PAYMENT
  // ---------------------------------------------------

  if (digitalRead(RENT_SWITCH) == LOW) {

    if (now - lastWarningBeep >= 2000UL) {

      lastWarningBeep = now;

      beep(1);

      // Show warning
      tft.fillRect(0, 15, 127, 10, BLACK);

      tft.setTextColor(RED);
      tft.setTextSize(1);

      tft.setCursor(18, 15);
      tft.print("PLEASE WAIT...");
    }
  }

  // ---------------------------------------------------
  // PAYMENT FINISHED
  // ---------------------------------------------------

  if (now - stateStartTime >= PAYMENT_TIME) {

    // -------------------------------------------------
    // SWITCH OFF -> READY
    // -------------------------------------------------

    if (digitalRead(RENT_SWITCH) == HIGH) {

      relayOFF();

      state = STATE_READY;
      stateStartTime = now;

      drawReady();

      return;
    }

    // -------------------------------------------------
    // SWITCH STILL ON -> WAIT
    // -------------------------------------------------

    state = STATE_WAIT;

    stateStartTime = now;

    drawWait();
  }
}

// =====================================================
// WAIT HANDLER
// =====================================================

void handleWait(unsigned long now) {

  // ===================================================
  // RELAY OFF
  // ===================================================

  relayOFF();

  drawWaitCountdown();

  // ---------------------------------------------------
  // BEEP WHILE SWITCH IS ON
  // ---------------------------------------------------

  if (
    digitalRead(RENT_SWITCH) == LOW &&
    now - lastWarningBeep >= 2000UL
  ) {

    lastWarningBeep = now;

    beep(1);
  }

  // ---------------------------------------------------
  // WAIT FINISHED
  // ---------------------------------------------------

  if (now - stateStartTime >= WAIT_TIME) {

    // -------------------------------------------------
    // SWITCH OFF
    // -------------------------------------------------

    if (digitalRead(RENT_SWITCH) == HIGH) {

      relayOFF();

      state = STATE_READY;

      stateStartTime = now;

      drawReady();
    }

    // -------------------------------------------------
    // SWITCH STILL ON
    // -------------------------------------------------

    else {

      // Restart another 15 second lock
      stateStartTime = now;

      drawWait();
    }
  }
}

// =====================================================
// ERROR HANDLER
// =====================================================

void handleError(unsigned long now) {

  // ===================================================
  // SAFETY:
  // RELAY MUST BE OFF
  // ===================================================

  relayOFF();

  if (now - stateStartTime >= 5000UL) {

    if (digitalRead(RENT_SWITCH) == HIGH) {

      state = STATE_READY;

      stateStartTime = now;

      drawReady();
    }
  }
}

// =====================================================
// BUZZER
// =====================================================

void beep(byte count) {

  for (byte i = 0; i < count; i++) {

    digitalWrite(BUZZER_PIN, HIGH);
    delay(100);

    digitalWrite(BUZZER_PIN, LOW);
    delay(100);
  }
}

// =====================================================
// DUMMY QR
// =====================================================

void drawDummyQR(int x, int y) {

  const int module = 3;
  const int size = 21;

  // ---------------------------------------------------
  // WHITE BACKGROUND
  // ---------------------------------------------------

  tft.fillRect(
    x,
    y,
    size * module,
    size * module,
    WHITE
  );

  // ---------------------------------------------------
  // QR PATTERN
  // ---------------------------------------------------

  for (int row = 0; row < size; row++) {

    for (int col = 0; col < size; col++) {

      bool blackPixel = false;

      // ------------------------------------------------
      // FINDER PATTERNS
      // ------------------------------------------------

      if (
        (row < 7 && col < 7) ||
        (row < 7 && col >= size - 7) ||
        (row >= size - 7 && col < 7)
      ) {

        int fx;
        int fy;

        // X coordinate
        if (col < 7) {
          fx = col;
        }
        else {
          fx = col - (size - 7);
        }

        // Y coordinate
        if (row < 7) {
          fy = row;
        }
        else {
          fy = row - (size - 7);
        }

        if (
          fx == 0 || fx == 6 ||
          fy == 0 || fy == 6 ||
          (fx >= 2 && fx <= 4 &&
           fy >= 2 && fy <= 4)
        ) {

          blackPixel = true;
        }
      }

      // ------------------------------------------------
      // PSEUDO QR DATA
      // ------------------------------------------------

      else {

        blackPixel =
          ((row * 13 +
            col * 7 +
            row * col) % 5) < 2;
      }

      // ------------------------------------------------
      // DRAW MODULE
      // ------------------------------------------------

      if (blackPixel) {

        tft.fillRect(
          x + col * module,
          y + row * module,
          module,
          module,
          BLACK
        );
      }
    }
  }
}
