// =====================================================

// Summary time
#define SUMMARY_TIME 15

// Payment time
#define PAYMENT_TIME 15

// Waiting/lock time
#define WAIT_TIME 15

// =====================================================
// TFT
// =====================================================

// =====================================================
// STATE MACHINE
// =====================================================

enum SystemState {
  WELCOME,
  READY,
  RENTING,
  SUMMARY,
  PAYMENT,
  WAIT_STATE,
  ERROR_STATE
};

SystemState state = WELCOME;

// VARIABLES
// =====================================================

float frequency = 0;

// =====================================================
// RELAY FUNCTIONS
// =====================================================

// Turn relay ON
void relayON() {
  digitalWrite(RELAY_PIN, LOW);
}

// Turn relay OFF
void relayOFF() {
  digitalWrite(RELAY_PIN, HIGH);
}

// =====================================================
// BUZZER
// =====================================================

void beep() {
  digitalWrite(BUZZER_PIN, HIGH);
  delay(100);
  digitalWrite(BUZZER_PIN, LOW);
}

void doubleBeep() {
  beep();
  delay(100);
  beep();
}

// =====================================================
// SETUP
// =====================================================

void setup() {

  // Serial monitor
  Serial.begin(115200);

  // PZEM
  Serial2.begin(9600);

  // Switch
  pinMode(RENT_SWITCH, INPUT_PULLUP);

  // Relay
  pinMode(RELAY_PIN, OUTPUT);

  // IMPORTANT:
  // Relay OFF at startup
  digitalWrite(RELAY_PIN, HIGH);

  // Buzzer
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  // TFT
  tft.initR(INITR_BLACKTAB);

  // Portrait mode = 128 x 160
  tft.setRotation(0);

  tft.fillScreen(BLACK);

  // PZEM
  pzem.begin();

  // Show welcome
  showWelcome();

  stateStartTime = millis();
}

// =====================================================
// LOOP
// =====================================================

void loop() {

  switch (state) {

    case WELCOME:
      handleWelcome();
      break;

    case READY:
      handleReady();
      break;

    case RENTING:
      handleRenting();
      break;

    case SUMMARY:
      handleSummary();
      break;

    case PAYMENT:
      handlePayment();
      break;

    case WAIT_STATE:
      handleWait();
      break;

    case ERROR_STATE:
      handleError();
      break;
  }
}

// =====================================================
// WELCOME HANDLER
// =====================================================

void handleWelcome() {

  unsigned long elapsed = millis() - stateStartTime;

  if (elapsed >= 5000) {

    state = READY;

    relayOFF();

    showReady();

    stateStartTime = millis();
  }
}

// =====================================================
// READ PZEM
// =====================================================

bool readPZEM() {

  float v = pzem.voltage();
  float c = pzem.current();
  float p = pzem.power();
  float e = pzem.energy();
  float f = pzem.frequency();
  float factor = pzem.pf();

  if (isnan(v) || isnan(c) || isnan(p)) {
    return false;
  }

  voltage = v;
  current = c;
  power = p;
  energy = e;
  frequency = f;
  pf = factor;

  return true;
}

// =====================================================
// READY HANDLER
// =====================================================

void handleReady() {

  // Relay must remain OFF
  relayOFF();

  if (millis() - lastReadTime >= 1000) {

    lastReadTime = millis();

    if (!readPZEM()) {
      showErrorMessage("PZEM ERROR");
      return;
    }

    showReady();
  }

  // Switch ON
  if (digitalRead(RENT_SWITCH) == LOW) {

    delay(80);

    if (digitalRead(RENT_SWITCH) == LOW) {

      startEnergy = energy;

      sessionEnergy = 0;
      sessionCost = 0;

      // =================================================
      // TURN RELAY ON
      // =================================================

      digitalWrite(RELAY_PIN, LOW);

      beep();

      state = RENTING;
      stateStartTime = millis();

      showRenting();

      delay(300);
    }
  }
}

// =====================================================
// RENTING HANDLER
// =====================================================

void handleRenting() {

  // Relay stays ON while renting
  digitalWrite(RELAY_PIN, LOW);

  // Read PZEM every second
  if (millis() - lastReadTime >= 1000) {

    lastReadTime = millis();

    if (!readPZEM()) {

      // Safety: turn relay OFF
      relayOFF();

      showErrorMessage("PZEM ERROR");

      state = ERROR_STATE;
      stateStartTime = millis();

      return;
    }

    // Calculate session energy
    if (energy >= startEnergy) {
      sessionEnergy = energy - startEnergy;
    }
    else {
      sessionEnergy = 0;
    }

    // Cost
    sessionCost = sessionEnergy * TARIFF;

    showRenting();
  }

  // Switch OFF
  if (digitalRead(RENT_SWITCH) == HIGH) {

    delay(80);

    if (digitalRead(RENT_SWITCH) == HIGH) {

      // =================================================
      // TURN RELAY OFF
      // =================================================

      digitalWrite(RELAY_PIN, HIGH);

      // Final calculation
      if (energy >= startEnergy) {
        sessionEnergy = energy - startEnergy;
      }
      else {
        sessionEnergy = 0;
      }

      sessionCost = sessionEnergy * TARIFF;

      beep();

      state = SUMMARY;
      stateStartTime = millis();

      showSummary();
    }
  }
}

// =====================================================
// SUMMARY HANDLER
// =====================================================

void handleSummary() {

  // Relay must remain OFF
  digitalWrite(RELAY_PIN, HIGH);

  unsigned long elapsed = millis() - stateStartTime;

  drawSummaryCountdown();

  // If switch is turned ON during summary
  if (digitalRead(RENT_SWITCH) == LOW) {

    beep();

    state = WAIT_STATE;
    stateStartTime = millis();

    showWait();

    return;
  }

  // Summary finished
  if (elapsed >= (SUMMARY_TIME * 1000UL)) {

    state = PAYMENT;
    stateStartTime = millis();

    showPayment();
  }
}

// =====================================================
// DUMMY QR CODE
// =====================================================

void drawDummyQR() {

  const int moduleSize = 3;

  const int qrSize = 21;

  const int qrPixelSize = qrSize * moduleSize;

  const int startX = (SCREEN_W - qrPixelSize) / 2;
  const int startY = 42;

  // White background
  tft.fillRect(
    startX - 2,
    startY - 2,
    qrPixelSize + 4,
    qrPixelSize + 4,
    WHITE
  );

  // Draw pseudo QR pattern
  for (int y = 0; y < qrSize; y++) {

    for (int x = 0; x < qrSize; x++) {

      bool black = false;

      // Finder patterns
      bool finder = false;

      // Top-left
      if (x < 7 && y < 7) {
        finder = true;

        if (
          x == 0 || x == 6 ||
          y == 0 || y == 6 ||
          (x >= 2 && x <= 4 &&
           y >= 2 && y <= 4)
        ) {
          black = true;
        }
      }

      // Top-right
      else if (x >= 14 && y < 7) {
        finder = true;

        int xx = x - 14;

        if (
          xx == 0 || xx == 6 ||
          y == 0 || y == 6 ||
          (xx >= 2 && xx <= 4 &&
           y >= 2 && y <= 4)
        ) {
          black = true;
        }
      }

      // Bottom-left
      else if (x < 7 && y >= 14) {
        finder = true;

        int yy = y - 14;

        if (
          x == 0 || x == 6 ||
          yy == 0 || yy == 6 ||
          (x >= 2 && x <= 4 &&
           yy >= 2 && yy <= 4)
        ) {
          black = true;
        }
      }

      if (!finder) {

        // Pseudo random pattern
        if (((x * 3 + y * 5 + x * y) % 7) < 3) {
          black = true;
        }
      }

      if (black) {

        tft.fillRect(
          startX + x * moduleSize,
          startY + y * moduleSize,
          moduleSize,
          moduleSize,
          BLACK
        );
      }
    }
  }
}

// =====================================================
// PAYMENT HANDLER
// =====================================================

void handlePayment() {

  // Relay MUST remain OFF
  digitalWrite(RELAY_PIN, HIGH);

  unsigned long elapsed = millis() - stateStartTime;

  drawPaymentCountdown();

  // If switch is turned ON during payment
  if (digitalRead(RENT_SWITCH) == LOW) {

    if (millis() - lastBeepTime >= 2000) {

      lastBeepTime = millis();

      beep();
    }

    tft.fillRect(0, 15, 127, 22, BLACK);

    tft.setTextColor(RED);
    tft.setTextSize(1);

    tft.setCursor(15, 15);
    tft.print("PLEASE WAIT...");
  }

  // Payment period finished
  if (elapsed >= (PAYMENT_TIME * 1000UL)) {

    // If switch is OFF, go READY
    if (digitalRead(RENT_SWITCH) == HIGH) {

      state = READY;
      stateStartTime = millis();

      showReady();

      return;
    }

    // If switch is still ON,
    // enter waiting state
    state = WAIT_STATE;
    stateStartTime = millis();

    showWait();
  }
}

// =====================================================
// WAIT HANDLER
// =====================================================

void handleWait() {

  // Relay MUST remain OFF
  digitalWrite(RELAY_PIN, HIGH);

  unsigned long elapsed = millis() - stateStartTime;

  int remaining = WAIT_TIME - (elapsed / 1000);

  if (remaining < 0) {
    remaining = 0;
  }

  tft.fillRect(35, 135, 60, 20, BLACK);

  tft.setTextColor(WHITE);
  tft.setTextSize(2);

  tft.setCursor(45, 135);

  if (remaining < 10) {
    tft.print("0");
  }

  tft.print(remaining);

  // Beep every 2 seconds if switch is still ON
  if (
    digitalRead(RENT_SWITCH) == LOW &&
    millis() - lastBeepTime >= 2000
  ) {

    lastBeepTime = millis();

    beep();
  }

  // Wait period finished
  if (elapsed >= (WAIT_TIME * 1000UL)) {

    // Only return to READY if switch is OFF
    if (digitalRead(RENT_SWITCH) == HIGH) {

      state = READY;
      stateStartTime = millis();

      showReady();
    }
    else {

      // Switch still ON
      stateStartTime = millis();

      showWait();
    }
  }
}

// =====================================================
// ERROR HANDLER
// =====================================================

void handleError() {

  // Safety first
  digitalWrite(RELAY_PIN, HIGH);

  unsigned long elapsed = millis() - stateStartTime;

    if (digitalRead(RENT_SWITCH) == HIGH) {

      state = READY;

      showReady();

      stateStartTime = millis();
    }
  }
}
